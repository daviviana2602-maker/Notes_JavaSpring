==================== LISTAS, SETS E MAPS ====================


// CRIAÇÃO

List<String> list = new ArrayList<>();

HashMap<String, Integer> map = new HashMap<>();

Set<Integer> set = new HashSet<>();


// LIST / ARRAYLIST

list.add("gato");
list.get(0);
list.set(0, "cachorro");
list.remove(0);
list.size();
list.contains("gato");
list.isEmpty();


// SET / HASHSET

set.add(10);
set.remove(10);
set.contains(10);
set.size();
set.isEmpty();
set.clear();


// HASHMAP

map.put("Davi", 18);
map.get("Davi");
map.remove("Davi");
map.containsKey("Davi");
map.containsValue(18);
map.size();
map.isEmpty();
map.clear();

==================== STREAM LISTA ====================


List<CartEntity> carrinho = cartRepository.findByOrderId(orderId);


return carrinho.stream()                        // passa por cada item da lista em sequência
    .map(item -> new CarrinhoResponse(          // transforma Entity em Response
        item.getId(),
        item.getPedidoId(),
        item.getProdutoId(),
        item.getQuantidade(),
        item.getPrecoUnit(),
        item.getPrecoTotal()
        ))
        .toList();                  // "chama" o stream e insere itens passados na lista
